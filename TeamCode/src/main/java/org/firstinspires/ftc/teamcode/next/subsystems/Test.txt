double relativeAngle=0;
double prevAbsAngle=0;

void updateCurrentAtStartOfLoop(){
double curAbsAngle=servo.getPosition();
double deltaAngle=0;
if(curAbsAngle<pi/2&&prevAbsAngle>3pi/2){
//wrapped around the positive side
deltaAngle= curAbsAngle+(2pi-prevAbsAngle);
}else if(curAbsAngle>3pi/2&&prevAbsAngle<pi/2){
//wrapped around the negative side
deltaAngle=-(prevAbsAngle+(2pi-curAbsAngle));
}else{
deltaAngle=curAbsAngle-prevAbsAngle;
}
relativeAngle+=deltaAngle;
prevAbsAngle=curAbsAngle;
}
